{{- /* tutorial-header shortcode — the tutorial's header banner. With `illustration: tutorial-…` in the front matter it draws that iso scene (assets/iso/.svg, exported from the design library) in the tutorial's product colour; otherwise it shows `header`, header-art.* or `src`. */ -}} {{- $alt := .Get "alt" | default .Page.Title -}} {{- with .Page.Params.illustration -}} {{- $svg := resources.Get (printf "iso/%s.svg" .) -}} {{- if not $svg -}} {{- errorf "tutorial-header: no assets/iso/%s.svg (illustration of %s)" . $.Page.RelPermalink -}} {{- end -}} {{- else -}} {{- $src := .Get "src" | default (.Page.Params.header | default "header-art.*") -}} {{- $image := .Page.Resources.GetMatch $src -}} {{- if not $image -}} {{- $image = resources.Get $src -}} {{- end -}} {{- with $image -}} {{- if eq .MediaType.SubType "svg" -}} {{- else -}} {{- $small := .Resize "800x webp q82" -}} {{- $large := .Resize "1200x webp q84" -}} {{- end -}} {{- else -}} {{- errorf "Tutorial header image %q not found for %s" $src .Page.RelPermalink -}} {{- end -}} {{- end -}}
